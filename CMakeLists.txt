cmake_minimum_required(VERSION 3.25)
project(PassageLab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(PASSAGELAB_DESKTOP "Build OpenGL desktop viewer" ON)
option(PASSAGELAB_TESTS "Build tests" ON)
find_package(Eigen3 CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(fcl CONFIG REQUIRED)
find_package(ompl CONFIG REQUIRED)
set(PL_DEPENDENCIES "{}")
if(EXISTS "${VCPKG_INSTALLED_DIR}/vcpkg/status")
  file(STRINGS "${VCPKG_INSTALLED_DIR}/vcpkg/status" PL_STATUS)
  foreach(PL_LINE IN LISTS PL_STATUS)
    if(PL_LINE MATCHES "^Package: (.+)$")
      set(PL_PACKAGE "${CMAKE_MATCH_1}")
    elseif(PL_LINE MATCHES "^Version: (.+)$")
      set(PL_PACKAGE_VERSION "${CMAKE_MATCH_1}")
      if(PL_PACKAGE MATCHES "^(eigen3|fcl|ccd|ompl|imgui|glfw3|glad|catch2|nlohmann-json|boost-headers)$")
        string(JSON PL_DEPENDENCIES SET "${PL_DEPENDENCIES}" "${PL_PACKAGE}" "\"${PL_PACKAGE_VERSION}\"")
      endif()
    endif()
  endforeach()
endif()
configure_file(cmake/dependency_versions.hpp.in generated/dependency_versions.hpp @ONLY)
execute_process(COMMAND git rev-parse --short HEAD WORKING_DIRECTORY ${CMAKE_SOURCE_DIR} OUTPUT_VARIABLE PL_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT PL_COMMIT)
  set(PL_COMMIT "uncommitted")
endif()
add_library(passagelab src/geometry.cpp src/planning.cpp src/ompl_adapter.cpp src/io.cpp src/scenes.cpp)
target_include_directories(passagelab PUBLIC include)
target_include_directories(passagelab PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(passagelab PUBLIC Eigen3::Eigen nlohmann_json::nlohmann_json fcl ompl::ompl bcrypt)
target_compile_definitions(passagelab PUBLIC PL_VERSION="${PROJECT_VERSION}" PL_COMMIT="${PL_COMMIT}" PL_BUILD="$<CONFIG>")
if(MSVC)
  target_compile_options(passagelab PRIVATE /W4 /permissive- /utf-8)
  target_compile_definitions(passagelab PUBLIC NOMINMAX _USE_MATH_DEFINES)
else()
  target_compile_options(passagelab PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(passagelab_cli apps/cli.cpp)
target_link_libraries(passagelab_cli PRIVATE passagelab)
install(TARGETS passagelab_cli RUNTIME DESTINATION .)
if(PASSAGELAB_DESKTOP)
  find_package(OpenGL REQUIRED)
  find_package(glfw3 CONFIG REQUIRED)
  find_package(glad CONFIG REQUIRED)
  find_package(imgui CONFIG REQUIRED)
  add_executable(passagelab_desktop apps/desktop.cpp)
  target_link_libraries(passagelab_desktop PRIVATE passagelab glfw glad::glad imgui::imgui OpenGL::GL)
  install(TARGETS passagelab_desktop RUNTIME DESTINATION .)
endif()
if(PASSAGELAB_TESTS)
  enable_testing()
  find_package(Catch2 3 CONFIG REQUIRED)
  add_executable(passagelab_tests tests/test_geometry.cpp tests/test_planning.cpp tests/test_io.cpp)
  target_link_libraries(passagelab_tests PRIVATE passagelab Catch2::Catch2WithMain)
  include(CTest)
  include(Catch)
  catch_discover_tests(passagelab_tests)
endif()
install(DIRECTORY assets/scenes DESTINATION assets)
install(FILES README.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
